To-do list

A small to-do list where you can add, complete and remove tasks. Your tasks are saved in the browser, so they are still there when you come back.

About the project

The page is built with HTML, styled with CSS and made interactive with JavaScript. Tasks are stored in the browser with localStorage, and text is added with textContent so anything typed into the field is shown as plain text and can't inject HTML into the page.

        <script>
            const form = document.querySelector("#todo-form");
            const input = document.querySelector("#todo-input");
            const list = document.querySelector("#todo-list");
            const count = document.querySelector("#todo-count");

            // Load saved tasks. Start empty if nothing is saved or the browser blocks storage.
            let tasks = [];
            try {
                tasks = JSON.parse(localStorage.getItem("tasks")) || [];
            } catch (error) {
                tasks = [];
            }
            if (!Array.isArray(tasks)) {
                tasks = [];
            }

            function save() {
                try {
                    localStorage.setItem("tasks", JSON.stringify(tasks));
                } catch (error) {
                    // Saving is optional, so the app keeps working without it
                }
            }

            function updateCount() {
                const left = tasks.filter((task) => !task.done).length;

                if (tasks.length === 0) {
                    count.textContent = "No tasks yet. Add one above.";
                } else {
                    count.textContent = left + (left === 1 ? " task left" : " tasks left");
                }
            }

            function renderTask(task) {
                const item = document.createElement("li");
                item.classList.toggle("done", task.done);

                const label = document.createElement("label");
                const checkbox = document.createElement("input");
                checkbox.type = "checkbox";
                checkbox.checked = task.done;

                // textContent (not innerHTML) shows typed text as plain text, so it can't inject HTML
                const text = document.createElement("span");
                text.textContent = task.text;
                label.append(checkbox, text);

                const remove = document.createElement("button");
                remove.type = "button";
                remove.className = "remove";
                remove.textContent = "Remove";
                remove.setAttribute("aria-label", "Remove task: " + task.text);

                checkbox.addEventListener("change", () => {
                    task.done = checkbox.checked;
                    item.classList.toggle("done", task.done);
                    save();
                    updateCount();
                });

                remove.addEventListener("click", () => {
                    tasks = tasks.filter((other) => other !== task);
                    item.remove();
                    save();
                    updateCount();
                    input.focus();
                });

                item.append(label, remove);
                list.append(item);
            }

            form.addEventListener("submit", (event) => {
                event.preventDefault();

                const text = input.value.trim();
                if (text === "") {
                    return;
                }

                const task = { text: text, done: false };
                tasks.push(task);
                renderTask(task);
                save();
                updateCount();

                input.value = "";
                input.focus();
            });
            tasks.forEach(renderTask);
            updateCount();
        </script>
